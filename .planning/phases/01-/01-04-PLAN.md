**Handler implementation:**

use axum::{extract::State, Json, http::StatusCode, response::IntoResponse};
use crate::channel_manager::ChannelManager;
use super::types::{CoinbaseUpdateRequest, CoinbaseUpdateResponse};

pub async fn handle_coinbase_update(
    State(channel_manager): State<ChannelManager>,
    Json(req): Json<CoinbaseUpdateRequest>,
) -> impl IntoResponse {
    // Phase 1: Validate request (API-03)
    if let Err(e) = req.validate() {
        return (StatusCode::BAD_REQUEST, Json(CoinbaseUpdateResponse::error(e)));
    }

    // Phase 2: Call ChannelManager update (validates address - API-02)
    match channel_manager.update_coinbase_and_broadcast(&req.address, req.user_id).await {
        Ok(_) => (StatusCode::OK, Json(CoinbaseUpdateResponse::success("Coinbase updated"))),
        Err(e) => {
            // Check error type to determine status code
            let status = if e.to_string().contains("Invalid") {
                StatusCode::BAD_REQUEST  // Address validation failed
            } else {
                StatusCode::INTERNAL_SERVER_ERROR  // Pool error
            };
            (status, Json(CoinbaseUpdateResponse::error(e.to_string())))
        }
    }
}
```

pub async fn start_api_server(
    channel_manager: ChannelManager,
    bind_addr: &str,
) -> Result<(), Box<dyn std::error::Error>> {
    let addr: SocketAddr = bind_addr.parse()?;
    let router = create_router(channel_manager);
    let listener = tokio::net::TcpListener::bind(addr).await?;

    tracing::info!("API server listening on {}", addr);
    axum::serve(listener, router).await?;
    Ok(())
}
```
